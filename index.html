""" CALCULATRICE SCIENTIFIQUE AVEC TRACEUR DE COURBES (Python) - VERSION MOBILE ============================================================================= Meme calculatrice que la version PC, adaptee pour un ecran de telephone : fenetre ajustee a l'ecran, boutons plus grands pour le tactile. INSTALLATION SUR ANDROID (via l'app Pydroid 3) : 1. Ouvre Pydroid 3 -> menu -> Pip -> installe "matplotlib" et "numpy" 2. Ouvre ce fichier dans Pydroid 3 3. Appuie sur le bouton Play (Run) INSTALLATION SUR PC : pip install matplotlib numpy python calculatrice_scientifique_mobile.py """ import tkinter as tk from tkinter import ttk, messagebox import math import matplotlib matplotlib.use("TkAgg") from matplotlib.figure import Figure from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg import numpy as np # Fonctions et constantes autorisees dans les expressions. # On limite volontairement ce qui est accessible (pas d'acces a des # fonctions systeme) pour eviter qu'une expression tapee par erreur # ne fasse quelque chose de dangereux. CONTEXTE_SUR = { "sin": math.sin, "cos": math.cos, "tan": math.tan, "asin": math.asin, "acos": math.acos, "atan": math.atan, "sqrt": math.sqrt, "log": math.log10, "ln": math.log, "exp": math.exp, "abs": abs, "pow": pow, "pi": math.pi, "e": math.e, } def evaluer_expression(expression, valeur_x=None): """Evalue une expression mathematique en toute securite (sans acces aux fonctions systeme), avec x optionnel pour le trace de courbes.""" contexte = dict(CONTEXTE_SUR) if valeur_x is not None: contexte["x"] = valeur_x expression = expression.replace("^", "**") # notation puissance plus naturelle # __builtins__ vide = aucune fonction Python standard accessible (securite) return eval(expression, {"__builtins__": {}}, contexte) class CalculatriceApp: def __init__(self, fenetre): self.fenetre = fenetre fenetre.title("Calculatrice scientifique") # Sur telephone (Pydroid 3), la fenetre doit occuper tout l'ecran # plutot qu'une taille fixe pensee pour un moniteur PC. try: fenetre.attributes("-zoomed", True) # plein ecran sur certains systemes except tk.TclError: largeur = fenetre.winfo_screenwidth() hauteur = fenetre.winfo_screenheight() fenetre.geometry(f"{largeur}x{hauteur}") conteneur = ttk.Notebook(fenetre) conteneur.pack(fill="both", expand=True) self.onglet_calcul = ttk.Frame(conteneur) self.onglet_graphique = ttk.Frame(conteneur) conteneur.add(self.onglet_calcul, text="Calculatrice") conteneur.add(self.onglet_graphique, text="Traceur de courbes") self._construire_onglet_calcul() self._construire_onglet_graphique() # ---------- ONGLET CALCULATRICE ---------- def _construire_onglet_calcul(self): cadre = self.onglet_calcul self.affichage = tk.Entry(cadre, font=("Consolas", 24), justify="right") self.affichage.grid(row=0, column=0, columnspan=5, sticky="nsew", padx=5, pady=10, ipady=16) boutons = [ ("sin(", 1, 0), ("cos(", 1, 1), ("tan(", 1, 2), ("sqrt(", 1, 3), ("(", 1, 4), ("log(", 2, 0), ("ln(", 2, 1), ("^", 2, 2), ("pi", 2, 3), (")", 2, 4), ("7", 3, 0), ("8", 3, 1), ("9", 3, 2), ("/", 3, 3), ("C", 3, 4), ("4", 4, 0), ("5", 4, 1), ("6", 4, 2), ("*", 4, 3), ("<-", 4, 4), ("1", 5, 0), ("2", 5, 1), ("3", 5, 2), ("-", 5, 3), ("=", 5, 4), ("0", 6, 0), (".", 6, 1), ("e", 6, 2), ("+", 6, 3), ] for (texte, ligne, colonne) in boutons: action = self._action_bouton(texte) # Boutons plus grands (police et hauteur) pour etre faciles a # toucher du doigt sur un ecran de telephone. tk.Button(cadre, text=texte, font=("Consolas", 18), width=5, height=3, command=action).grid(row=ligne, column=colonne, padx=3, pady=3, sticky="nsew") for i in range(5): cadre.grid_columnconfigure(i, weight=1) for i in range(1, 7): cadre.grid_rowconfigure(i, weight=1) def _action_bouton(self, texte): def action(): if texte == "C": self.affichage.delete(0, tk.END) elif texte == "<-": contenu = self.affichage.get() self.affichage.delete(0, tk.END) self.affichage.insert(0, contenu[:-1]) elif texte == "=": self._calculer() else: self.affichage.insert(tk.END, texte) return action def _calculer(self): expression = self.affichage.get() try: resultat = evaluer_expression(expression) self.affichage.delete(0, tk.END) self.affichage.insert(0, str(resultat)) except Exception: messagebox.showerror("Erreur", "Expression invalide.") # ---------- ONGLET TRACEUR DE COURBES ---------- def _construire_onglet_graphique(self): cadre = self.onglet_graphique # Deux rangees empilees (plutot qu'une seule ligne large) pour que # tout reste visible sur la largeur etroite d'un ecran de telephone. rangee1 = ttk.Frame(cadre) rangee1.pack(fill="x", padx=10, pady=(10, 5)) ttk.Label(rangee1, text="f(x) =", font=("Consolas", 12)).pack(side="left") self.champ_fonction = ttk.Entry(rangee1, font=("Consolas", 14)) self.champ_fonction.insert(0, "sin(x) + x^2/10") self.champ_fonction.pack(side="left", padx=5, fill="x", expand=True) rangee2 = ttk.Frame(cadre) rangee2.pack(fill="x", padx=10, pady=(0, 10)) ttk.Label(rangee2, text="x min").pack(side="left") self.champ_xmin = ttk.Entry(rangee2, width=6) self.champ_xmin.insert(0, "-10") self.champ_xmin.pack(side="left", padx=(2, 10)) ttk.Label(rangee2, text="x max").pack(side="left") self.champ_xmax = ttk.Entry(rangee2, width=6) self.champ_xmax.insert(0, "10") self.champ_xmax.pack(side="left", padx=(2, 10)) ttk.Button(rangee2, text="Tracer", command=self._tracer_courbe).pack(side="left", padx=5) self.figure = Figure(figsize=(7, 4.5), dpi=100) self.axes = self.figure.add_subplot(111) self.canevas = FigureCanvasTkAgg(self.figure, master=cadre) self.canevas.get_tk_widget().pack(fill="both", expand=True, padx=10, pady=10) def _tracer_courbe(self): try: xmin = float(self.champ_xmin.get()) xmax = float(self.champ_xmax.get()) expression = self.champ_fonction.get() x = np.linspace(xmin, xmax, 500) y = [evaluer_expression(expression, valeur_x=xi) for xi in x] self.axes.clear() self.axes.plot(x, y, color="#1f4e79", linewidth=2) self.axes.axhline(0, color="gray", linewidth=0.5) self.axes.axvline(0, color="gray", linewidth=0.5) self.axes.set_title(f"f(x) = {expression}") self.axes.grid(True, linestyle="--", alpha=0.5) self.canevas.draw() except Exception as e: messagebox.showerror("Erreur", f"Impossible de tracer cette fonction.\n{e}") def main(): fenetre = tk.Tk() CalculatriceApp(fenetre) fenetre.mainloop() if __name__ == "__main__": main()
